create table developpeur (
	NumDev char(4),
	NomDev varchar(30) not null,
	AdrDev varchar(30) not null,
	EmailDev varchar(40),
	TelDev char(8),
	constraint NumDev_ck check ( NumDev like ('Dv%')) ,
	constraint NomDev_ck check ( NomDev like ('%@%')) 
);

alter table developpeur add constraint numdev_pk primary key (NumDev);

alter table developpeur drop constraint NomDev_ck;

alter table developpeur add constraint EmailDev_ck check ( EmailDev like ('%@%'));

create table projet (
	NumProj char(4) primary key,
	TitreProj varchar(30) not null,
	DateDeb date
);

create table logiciel(
	CodLog char(4) primary key ,
	NomLog varchar(30) not null,
	PrixLog int check (PrixLog >800),
	NumProj char(4),
	constraint numproj_fk foreign key (NumProj) references projet (NumProj)
);

create table realisation (
	NumProj char(4),
	NumDev char(4),
	constraint num_pk primary key (NumProj,NumDev),
	constraint numProj_fk2 foreign key (NumProj) references projet (NumProj),
	constraint numDev_fk foreign key (NumDev) references developpeur (NumDev)
);
