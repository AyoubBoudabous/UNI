create table departement(
	NumDept char(6) primary key,
	Appellation varchar(6),
	Constraint app_ck check (Appellation in ('BIO','SCENV','N-TECH'))
);
	
create table employe (
	Matricule char(6) primary key,
	Nom varchar(20) not null,
	Prenom varchar(20) not null,
	Date_N date,
	Depart char(6),
	Adresse varchar(10) check (Adresse in ('Paris', 'Rennes', 'Marseille', 'Lyon', 'Strasbourg', 'Bordeaux', 'Lille', 'Toulouse', 'Toulon')),
	sal_brut int check (sal_brut > 1000),
	charge_sociale int default 0,
	prime int default 0,
	retenue int default 0
);

 alter table employe
 add constraint depart_fk foreign key (Depart) references departement (NumDept);

create table Projet (
	Numero char(4) primary key,
	Designation varchar(10),
	Budget int,
	date_debut date,
	date_fin date,
	constraint date_ck check (date_debut < date_fin)
);


select ASCII('é') from dual;

create table Role (
	NumE char(6) ,
	NumP char(4) ,
	Role_Emp varchar(20) check (Role_Emp in ('Responsable', 'Ing'||chr(14844058)||'nieur', 'Technicien')),
	constraint num_pk primary key (NumE,NumP),
	constraint numE_fk foreign key (NumE) references employe (Matricule),
	constraint numP_fk foreign key (NumP) references Projet (Numero)
);
